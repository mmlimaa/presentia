    login = models.CharField(max_length=50, unique=True)
    senha_hash = models.CharField(max_length=255)
    email = models.EmailField(max_length=100, unique=True)
    telefone = models.CharField(max_length=15)
    is_admin = models.BooleanField(default=False)
    data_cadastro = models.DateTimeField(auto_now_add=True)

    class Meta:
        db_table = 'tb_usuarios'

    def __str__(self):

class SolicitacaoAcessoForm(forms.ModelForm):
    nome_completo = forms.CharField(max_length=150, required=True, widget=forms.TextInput(attrs={'class': 'w-full px-4 py-2 border rounded-lg', 'placeholder': 'NOME COMPLETO'}))
    login = forms.CharField(max_length=30, required=True, widget=forms.TextInput(attrs={'class': 'w-full px-4 py-2 border rounded-lg', 'placeholder': 'login'}))
    senha = forms.CharField(widget=forms.PasswordInput(attrs={'class': 'w-full px-4 py-2 border rounded-lg', 'placeholder': 'Senha (6 a 16 caracteres)'}), min_length=6, max_length=16, required=True)
    email = forms.EmailField(max_length=100, required=True, widget=forms.EmailInput(attrs={'class': 'w-full px-4 py-2 border rounded-lg', 'placeholder': 'email@exemplo.com'}))
    telefone = forms.CharField(max_length=15, required=True, widget=forms.TextInput(attrs={'class': 'w-full px-4 py-2 border rounded-lg', 'placeholder': '(11) 99999-9999'}))

    class Meta:
        model = SolicitacaoAcesso
        fields = ['nome_completo', 'login', 'email', 'telefone']

    def clean_nome_completo(self):
        nome = self.cleaned_data.get('nome_completo')
        return nome.upper() if nome else nome

    def clean_login(self):
        login = self.cleaned_data.get('login')
        if login:
            login = login.lower()
            if Usuario.objects.filter(login=login).exists() or SolicitacaoAcesso.objects.filter(login=login).exists():
                raise forms.ValidationError("Este login já está em uso ou em análise.")
        return login

    def clean_email(self):
        email = self.cleaned_data.get('email')
        if email:
            email = email.lower()
            if Usuario.objects.filter(email=email).exists() or SolicitacaoAcesso.objects.filter(email=email).exists():
                raise forms.ValidationError("Este e-mail já está cadastrado ou em análise.")
        return email

    def save(self, commit=True):
        solicitacao = super().save(commit=False)
        solicitacao.senha_hash = make_password(self.cleaned_data['senha'])
        if commit:
            solicitacao.save()
        return solicitacao

def login_view(request):
    if request.session.get('usuario_id'):
        return redirect('dashboard')

    if request.method == 'POST':
        login_digitado = request.POST.get('username', '').strip().lower()
        senha_digitada = request.POST.get('password', '').strip()

        if not login_digitado or not senha_digitada:
            messages.error(request, "Por favor, preencha todos os campos.")
            return render(request, 'login.html')

        try:
            user = Usuario.objects.get(login=login_digitado)
            if user.check_senha(senha_digitada):
                request.session['usuario_id'] = user.id
                request.session['usuario_nome'] = user.nome_completo
                request.session['usuario_is_admin'] = user.is_admin
                return redirect('dashboard')
            else:
                messages.error(request, "Senha incorreta.")
        except Usuario.DoesNotExist:
            messages.error(request, "Usuário não encontrado.")
            
    return render(request, 'login.html')

def logout_view(request):
    request.session.flush()
    return redirect('login')

def alunos_nucleo(request, nucleo_id):
    if not request.session.get('usuario_id'):
        return redirect('login')
    
    nucleo = get_object_or_404(Nucleo, id=nucleo_id)
    cursos = Curso.objects.all().order_by('ordem')
    
    # Se nenhum curso foi selecionado, pega o primeiro por padrao
    curso_id = request.GET.get('curso')
    if curso_id:
        curso_selecionado = get_object_or_404(Curso, id=curso_id)
    else:
        curso_selecionado = cursos.first()
        
    if curso_selecionado:
        matriculas = Matricula.objects.filter(nucleo=nucleo, curso=curso_selecionado).select_related('aluno').order_by('aluno__nome')
        experimentais = AlunoExperimental.objects.filter(nucleo=nucleo, curso=curso_selecionado).order_by('-data_visita')
    else:
        matriculas = Matricula.objects.none()
        experimentais = AlunoExperimental.objects.none()

def frequencia_turma(request, turma_id):
    if not request.session.get('usuario_id'):
        return redirect('login')
    
    turma = get_object_or_404(Turma, id=turma_id)
    data_str = request.GET.get('data')
    
    from datetime import datetime
    if not data_str:
        data_str = datetime.today().strftime('%Y-%m-%d')
        
    try:
        data_aula = datetime.strptime(data_str, '%Y-%m-%d').date()
    except ValueError:
        data_aula = datetime.today().date()
        data_str = data_aula.strftime('%Y-%m-%d')

    from presentia.attendance.models import Aula as AulaModel, Frequencia as FreqModel
    
    # 1. Busca ou cria a Aula
    aula = AulaModel.objects.filter(nucleo=turma.nucleo, turma=turma, data=data_aula).first()
    if not aula:
        aula = AulaModel.objects.create(nucleo=turma.nucleo, turma=turma, data=data_aula, topico='Aula Regular', curso=turma.curso)
    
    # 2. Busca as frequencias existentes
    frequencias = FreqModel.objects.filter(aula=aula).select_related('matricula__aluno').order_by('matricula__aluno__nome')
    
    # 3. Se nao existirem, cria para os alunos ativos
    if not frequencias.exists():
        matriculas_ativas = Matricula.objects.filter(turma=turma, ativo=True).order_by('aluno__nome')
        novas_frequencias = [
            FreqModel(aula=aula, matricula=mat, status=FreqModel.Status.AUSENTE) 
            for mat in matriculas_ativas
        ]
        FreqModel.objects.bulk_create(novas_frequencias)
        frequencias = FreqModel.objects.filter(aula=aula).select_related('matricula__aluno').order_by('matricula__aluno__nome')

    # 4. Processa o POST
    if request.method == 'POST':
        for freq in frequencias:
            novo_status = request.POST.get(f'status_{freq.id}')
            if novo_status in [FreqModel.Status.PRESENTE, FreqModel.Status.AUSENTE, FreqModel.Status.JUSTIFICADA]:
                freq.status = novo_status
                freq.save()
        messages.success(request, "Chamada salva com sucesso!")
        return redirect(f"{request.path}?data={data_str}")

    # 5. Calcula % Frequencia (Presencas / Total de Aulas Lancadas)
    total_aulas_turma = AulaModel.objects.filter(turma=turma).count()
    
    for freq in frequencias:
        if total_aulas_turma > 0:
            presencas_aluno = FreqModel.objects.filter(
                matricula=freq.matricula, 
                status=FreqModel.Status.PRESENTE
            ).count()
            freq.percentual_frequencia = int((presencas_aluno / total_aulas_turma) * 100)
        else:
            freq.percentual_frequencia = 0

    # 6. Contadores
    total_alunos = frequencias.count()
    total_presentes = frequencias.filter(status=FreqModel.Status.PRESENTE).count()
    total_ausentes = frequencias.filter(status=FreqModel.Status.AUSENTE).count()
    total_justificadas = frequencias.filter(status=FreqModel.Status.JUSTIFICADA).count()

    # 7. Busca alunos experimentais
    experimentais = AlunoExperimental.objects.filter(nucleo=turma.nucleo, curso=turma.curso, data_visita__date=data_aula).order_by('nome_completo')
    
    # 8. Datas com frequencia para o calendario
    datas_com_frequencia = AulaModel.objects.filter(turma=turma).values_list('data', flat=True)
    datas_iso = [d.strftime('%Y-%m-%d') for d in datas_com_frequencia]
    
    return render(request, 'frequencia_turma.html', {
        'turma': turma,
        'frequencias': frequencias,
        'experimentais': experimentais,
        'data_str': data_str,
        'datas_com_frequencia': datas_iso,
        'total_alunos': total_alunos,
        'total_presentes': total_presentes,
        'total_ausentes': total_ausentes,
        'total_justificadas': total_justificadas,
    })

                SELECT t.id, t.nome, t.data_inicio, t.data_fim, c.nome as curso_nome
                FROM tb_turmas t
                JOIN tb_cursos c ON c.id = t.curso_id
                WHERE t.nucleo_id = %s::uuid
                ORDER BY t.data_inicio DESC
            ''', [nucleo_id])

        for row in rows:
            turma_id = str(row[0])

                cursor.execute("SELECT COUNT(*) FROM tb_matriculas WHERE turma_id = %s::uuid AND ativo = true", [turma_id])

                cursor.execute("SELECT COUNT(*) FROM tb_matriculas WHERE turma_id = %s::uuid AND ativo = false", [turma_id])

def consulta_frequencia(request):
    if not request.session.get('usuario_id'):
        return redirect('login')
    
    from presentia.churches.models import Igreja
    from presentia.courses.models import Nucleo, Turma
    
    igrejas = Igreja.objects.all().order_by('nome')
    nucleos = Nucleo.objects.all().order_by('nome')
    
    igreja_id = request.GET.get('igreja')
    nucleo_id = request.GET.get('nucleo')
    turma_id = request.GET.get('turma')
    aluno_nome = request.GET.get('aluno_nome')
    
    if igreja_id:
        nucleos = nucleos.filter(igreja_id=igreja_id)
        
    turmas = Turma.objects.none()
    if nucleo_id:
        turmas = Turma.objects.filter(nucleo_id=nucleo_id).order_by('-data_inicio')

    contexto = {
        'igrejas': igrejas, 
        'nucleos': nucleos, 
        'turmas': turmas,
        'igreja_id': igreja_id, 
        'nucleo_id': nucleo_id,
        'turma_id': turma_id,
        'aluno_nome': aluno_nome,
    }

    if turma_id:
        try:
            turma = get_object_or_404(Turma, id=turma_id)
            data_inicio = turma.data_inicio
            data_fim = turma.data_fim

            frequencias = Frequencia.objects.filter(
                aula__turma=turma,
                aula__data__gte=data_inicio,
                aula__data__lte=data_fim
            ).select_related('matricula__aluno', 'aula')

            if aluno_nome:
                frequencias = frequencias.filter(matricula__aluno__nome__icontains=aluno_nome)

            agrupado = {}
            for f in frequencias:
                mes_ano = f.aula.data.strftime('%Y-%m')
                nome = f.matricula.aluno.nome
                data_str = f.aula.data.strftime('%Y-%m-%d')
                
                if mes_ano not in agrupado:
                    agrupado[mes_ano] = {}
                if nome not in agrupado[mes_ano]:
                    agrupado[mes_ano][nome] = {}
                
                agrupado[mes_ano][nome][data_str] = f.status

            meses_pt = {'01': 'JAN', '02': 'FEV', '03': 'MAR', '04': 'ABR', '05': 'MAI', '06': 'JUN',
                        '07': 'JUL', '08': 'AGO', '09': 'SET', '10': 'OUT', '11': 'NOV', '12': 'DEZ'}

            blocos = []
            for mes in sorted(agrupado.keys()):
                aulas_mes = Aula.objects.filter(
                    turma=turma, 
                    data__year=int(mes.split('-')[0]), 
                    data__month=int(mes.split('-')[1])
                ).order_by('data')
                dias_do_mes = [a.data.day for a in aulas_mes]

                ano_mes = mes.split('-')
                mes_formatado = f"{meses_pt[ano_mes[1]]}/{ano_mes[0]}"

                alunos_do_mes = []
                for nome in sorted(agrupado[mes].keys()):
                    aluno_linha = {'nome': nome, 'dias': []}
                    for dia in dias_do_mes:
                        chave_data = f"{mes}-{dia:02d}"
                        status = agrupado[mes][nome].get(chave_data)
                        
                        if status == 'presente':
                            aluno_linha['dias'].append({'dia': dia, 'status': 'P', 'cor': 'text-green-600'})
                        elif status == 'ausente':
                            aluno_linha['dias'].append({'dia': dia, 'status': 'A', 'cor': 'text-red-600'})
                        elif status == 'justificada':
                            aluno_linha['dias'].append({'dia': dia, 'status': 'J', 'cor': 'text-black'})
                        else:
                            aluno_linha['dias'].append({'dia': dia, 'status': '-', 'cor': 'text-gray-400'})
                    
                    alunos_do_mes.append(aluno_linha)
                
                blocos.append({
                    'mes_ano': mes_formatado,
                    'dias': dias_do_mes,
                    'alunos': alunos_do_mes
                })

            contexto.update({
                'turma': turma,
                'blocos': blocos, 
                'gerado': True,
                'data_ini_formatada': data_inicio.strftime('%d/%m/%Y'),
                'data_fim_formatada': data_fim.strftime('%d/%m/%Y')
            })
        except Exception as e:
            messages.error(request, f"Erro no relatÃ³rio: {e}")

    if request.GET.get('export') == 'pdf' and contexto.get('gerado'):
        return gerar_pdf_relatorio(contexto)

    return render(request, 'consulta_frequencia.html', contexto)

def importar_planilha_financeira(request):
    if not request.session.get('usuario_id') or not request.session.get('usuario_is_admin'):
        return redirect('login')
    
    turmas = Turma.objects.all().order_by('-data_inicio')

    if request.method == 'POST':
        turma_id = request.POST.get('turma')
        arquivo = request.FILES.get('planilha')
        
        if not turma_id or not arquivo:
            messages.error(request, "Selecione a turma e envie o arquivo CSV.")
            return redirect('importar_planilha_financeira')

        try:
            turma = Turma.objects.get(id=turma_id)
            data = arquivo.read().decode('utf-8-sig')
            leitor = csv.reader(io.StringIO(data), delimiter=',')
            linhas = list(leitor)
            
            if len(linhas) > 0 and "Nome" in linhas[0][0]:
                linhas = linhas[1:]
                
            importados = 0
            alunos_cancelados = 0
            secao_atual = 'ativo'

            for linha in linhas:
                if not linha or len(linha) < 2: 
                    continue
                
                linha_texto = " ".join(linha).upper()
                
                if "CANCELADO" in linha_texto:
                    if "CLIENTE" in linha_texto:
                        secao_atual = 'cancelado_cliente'
                    elif "INADIMPLENTE" in linha_texto:
                        secao_atual = 'cancelado_inadimplente'
                    else:
                        secao_atual = 'cancelado_generico'
                    continue

                email = ""
                for celula in linha:
                    if "@" in celula and "." in celula:
                        email = celula.strip().lower()
                        break
                
                if not email:
                    continue

                nome = linha[0].strip().upper() if len(linha) > 0 else "ALUNO SEM NOME"
                telefone = linha[1].strip().replace(" ", "").replace("-", "") if len(linha) > 1 else ""

                aluno, _ = Aluno.objects.get_or_create(
                    email=email, 
                    defaults={'nome': nome, 'telefone': telefone, 'data_inicio': datetime.today().date()}
                )
                
                matricula, _ = Matricula.objects.get_or_create(
                    aluno=aluno, turma=turma,
                    defaults={'nucleo': turma.nucleo, 'curso': turma.curso}
                )
                
                Mensalidade.objects.filter(matricula=matricula).delete()
                status_geral = 'Em dia'
                
                if secao_atual.startswith('cancelado'):
                    matricula.ativo = False
                    if secao_atual == 'cancelado_inadimplente':
                        status_geral = 'Inadimplente'
                    alunos_cancelados += 1
                else:
                    matricula.ativo = True
                    for i in range(4, len(linha)):
                        valor = linha[i].strip().upper()
                        if not valor: continue
                        numero_mes = i - 3
                        
                        if 'PAGO' in valor:
                            status = 'PAGO'
                        elif 'INADIMPLENTE' in valor:
                            status = 'INADIMPLENTE'
                            status_geral = 'Inadimplente'
                        else:
                            continue
                            
                        data_status = None
                        match = re.search(r'(\d{2}/\d{2}/\d{4})', valor)
                        if match:
                            try:
                                data_status = datetime.strptime(match.group(1), '%d/%m/%Y').date()
                            except ValueError:
                                pass
                        
                        Mensalidade.objects.create(
                            matricula=matricula,
                            numero_mes=numero_mes,
                            status=status,
                            data_status=data_status
                        )
                    
                    if not Mensalidade.objects.filter(matricula=matricula).exists():
                        status_geral = 'Quitado'

                matricula.status_pagamento = status_geral
                matricula.save()
                importados += 1

            messages.success(request, f"{importados} alunos processados ({alunos_cancelados} desativados).")
            return redirect('alunos_turma', turma_id=turma.id)
            
        except Exception as e:
            messages.error(request, f"Erro ao importar: {e}")
            return redirect('importar_planilha_financeira')

    return render(request, 'importar_financeiro.html', {'turmas': turmas})

        turmas_processar = Turma.objects.filter(id=turma_id)

        turmas_processar = Turma.objects.filter(nucleo_id=nucleo_id)
    else:
        turmas_processar = Turma.objects.filter(nucleo__igreja_id=igreja_id)

    # Agrupamento
    dados_agrupados = {}
    for turma in turmas_processar:
        igreja_nome = turma.nucleo.igreja.nome
        nucleo_nome = turma.nucleo.nome
        
        if igreja_nome not in dados_agrupados:
            dados_agrupados[igreja_nome] = {}
        if nucleo_nome not in dados_agrupados[igreja_nome]:
            dados_agrupados[igreja_nome][nucleo_nome] = []
            
        total_aulas = Aula.objects.filter(turma=turma).count()

        total_ativos = 0
        total_desligados = 0
        
        for mat in matriculas:

            if pagamento_filtro != 'todos':
                if pagamento_filtro == 'avista' and mat.status_pagamento != 'Quitado':
                    continue
                elif pagamento_filtro == 'adimplente' and mat.status_pagamento != 'Em dia':
                    continue
                elif pagamento_filtro == 'inadimplente' and mat.status_pagamento != 'Inadimplente':
                    continue

            if mat.ativo:
                total_ativos += 1
            else:
                total_desligados += 1
                
            percentual = 0
            if total_aulas > 0 and mat.ativo:
                presencas = Frequencia.objects.filter(matricula=mat, status=Frequencia.Status.PRESENTE).count()

        if alunos_dados:
            dados_agrupados[igreja_nome][nucleo_nome].append({
                'nome': turma.nome,
                'curso': turma.curso.nome,
                'total_ativos': total_ativos,
                'total_desligados': total_desligados,
                'alunos': alunos_dados
            })

    # Verifica se o usuario clicou em "Exibir" ou "Gerar PDF"
    exibir_dados = request.GET.get('exibir') == 'true' or request.GET.get('export') == 'pdf'
    
    contexto = {
        'igrejas': igrejas,
        'nucleos': nucleos,
        'turmas': turmas,
        'igreja_id': igreja_id,
        'nucleo_id': nucleo_id,
        'turma_id': turma_id,
        'pagamento_filtro': pagamento_filtro,
        'dados_agrupados': dados_agrupados if exibir_dados else {},
        'gerado': exibir_dados
    }

    if request.GET.get('export') == 'pdf' and contexto.get('gerado'):
        return gerar_pdf_relatorio_frequencia(contexto)

    return render(request, 'relatorio_frequencia_aulas.html', contexto)


def gerar_pdf_relatorio_frequencia(contexto):
    response = HttpResponse(content_type='application/pdf')
    response['Content-Disposition'] = 'attachment; filename="relatorio_frequencia_aulas.pdf"'
    
    doc = SimpleDocTemplate(response, pagesize=portrait(A4), rightMargin=30, leftMargin=30, topMargin=30, bottomMargin=30)
    elements = []
    
    styles = getSampleStyleSheet()
    title_style = ParagraphStyle(name='TitleStyle', parent=styles['Heading1'], alignment=1)
    sub_style = ParagraphStyle(name='SubStyle', parent=styles['Normal'], alignment=1)
    
    elements.append(Paragraph("RelatÃ³rio de FrequÃªncia das Aulas", title_style))
    elements.append(Paragraph("Presentia - Controle de FrequÃªncia", sub_style))
    elements.append(Spacer(1, 20))

    for igreja_nome, nucleos in contexto['dados_agrupados'].items():
        elements.append(Paragraph(f"Igreja: {igreja_nome}", ParagraphStyle(name='IgrejaStyle', parent=styles['Heading2'])))
        elements.append(Spacer(1, 10))
        
        for nucleo_nome, turmas in nucleos.items():
            elements.append(Paragraph(f"NÃºcleo: {nucleo_nome}", ParagraphStyle(name='NucleoStyle', parent=styles['Heading3'])))
            elements.append(Spacer(1, 10))
            
            for turma in turmas:
                elements.append(Paragraph(f"Turma: {turma['nome']} ({turma['curso']}) - Ativos: {turma['total_ativos']} | Desligados: {turma['total_desligados']}", ParagraphStyle(name='TurmaStyle', parent=styles['Normal'])))
                elements.append(Spacer(1, 5))
                
                # Tabela de Alunos
                header = ["Aluno", "Aproveitamento", "Pagamento", "Status"]
                data = [header]
                
                for aluno in turma['alunos']:
                    status = "Ativo" if aluno['ativo'] else "Desligado"
                    data.append([aluno['nome'], f"{aluno['aproveitamento']}%", aluno['status_pagamento'], status])
                    
                table = Table(data, colWidths=[250, 80, 80, 60])
                style = TableStyle([
                    ('BACKGROUND', (0, 0), (-1, 0), colors.HexColor('#1e293b')),
                    ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
                    ('ALIGN', (0, 0), (0, -1), 'LEFT'),
                    ('ALIGN', (1, 0), (-1, -1), 'CENTER'),
                    ('FONTNAME', (0, 0), (-1, 0), 'Helvetica-Bold'),
                    ('FONTSIZE', (0, 0), (-1, -1), 9),
                    ('BOTTOMPADDING', (0, 0), (-1, 0), 10),
                    ('BACKGROUND', (0, 1), (-1, -1), colors.white),
                    ('GRID', (0, 0), (-1, -1), 0.5, colors.grey),
                    ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
                ])
                
                # Colorir aproveitamento
                for r_idx, aluno in enumerate(turma['alunos'], start=1):
                    if aluno['aproveitamento'] >= 75:
                        style.add('TEXTCOLOR', (1, r_idx), (1, r_idx), colors.green)
                    elif aluno['aproveitamento'] >= 50:
                        style.add('TEXTCOLOR', (1, r_idx), (1, r_idx), colors.orange)
                    else:
                        style.add('TEXTCOLOR', (1, r_idx), (1, r_idx), colors.red)
                        
                    # Colorir status
                    if aluno['status_pagamento'] == 'Inadimplente':
                        style.add('TEXTCOLOR', (2, r_idx), (2, r_idx), colors.red)
                        
                table.setStyle(style)
                elements.append(table)
                elements.append(Spacer(1, 20))
    
    doc.build(elements)
    return response

def lista_nucleos(request):
    if not request.session.get('usuario_id'):
        return redirect('login')
        
    from django.db import connection
    from presentia.churches.models import Igreja
    
    # Pega todas as igrejas para o filtro
    igrejas = Igreja.objects.all().order_by('nome')
    
    # Pega o filtro da URL
    igreja_filtro = request.GET.get('igreja')
    
    query = '''
        SELECT n.id, n.nome, i.id as igreja_id, i.nome AS igreja_nome, i.cidade,
               (SELECT COUNT(*) FROM tb_matriculas m WHERE m.nucleo_id = n.id) AS total_alunos,
               (SELECT COUNT(*) FROM tb_turmas t WHERE t.nucleo_id = n.id) AS total_turmas
        FROM tb_nucleos n
        JOIN tb_igrejas i ON i.id = n.igreja_id
    '''
    params = []
    
    if igreja_filtro:
        query += " WHERE i.id = %s::uuid"
        params.append(igreja_filtro)
        
    query += " ORDER BY n.nome"
    
    with connection.cursor() as cursor:
        cursor.execute(query, params)
        rows = cursor.fetchall()

    nucleos = []
    for row in rows:
        nucleos.append({
            'id': row[0],
            'nome': row[1],
            'igreja_nome': row[3],
            'cidade': row[4],
            'total_alunos': row[5],
            'total_turmas': row[6],
        })

    return render(request, 'nucleos_lista.html', {
        'nucleos': nucleos, 
        'igrejas': igrejas,
        'igreja_filtro': igreja_filtro
    })
